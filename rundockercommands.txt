# 2. Build and start containers in background

# 3. Check status of both containers (should show Healthy & Running)
docker compose ps

# 4. View Backend logs
docker compose logs -f backend

# 5. Open in Browser
# http://localhost:5000/


# ------------------------------------------------------------------------------
# OTHER METHOD: STANDALONE DOCKER RUN
# ------------------------------------------------------------------------------
# 1. Navigate to backend directory
cd "C:\Users\Abcom\OneDrive\Desktop\FINAL YEAR PROJECT KUTUMBSETU\backend"

# 2. Build the Docker Image
docker build -t kutumbsetu-app .

# 3. Remove any previous container
docker rm -f kutumb-container

# 4. Run the Container (connected to MongoDB network)
docker run -d -p 5000:5000 --name kutumb-container --network finalyearprojectkutumbsetu_default -e MONGO_URI="mongodb://kutumbsetu-mongodb:27017/kutumbsetu" kutumbsetu-app

# 5. Check container logs
docker logs -f kutumb-container

# 6. Open in Browser
# http://localhost:5000/
